/**
 * EVIDENCE (not a fix): NOTIF-SEEN-01 says a parked-task notification is seen
 * once the task no longer waits. Requeueing a parked card (the push button
 * "Rimetti in coda" = PATCH status todo, shared/notify-actions.ts) leaves the
 * bell row unseen: only an exit from `review` clears `task:<id>` rows.
 */
import { describe, expect, test, beforeAll, afterAll } from "bun:test";
import { mkdtempSync, rmSync, mkdirSync, writeFileSync, readFileSync, readdirSync } from "fs";
import { tmpdir } from "os";
import { join } from "path";
import { initDatabase, closeDatabase } from "../db";
import { unseenSnapshot } from "../db/notification-log";
import { configureNotificationRegistry, recordAndAnnounce } from "../notification-registry";
import { createTaskService } from "./tasks";
import { freshDb } from "./tasks-test-db";
import { taskParkedNotificationKey, taskReviewNotificationKey } from "../../shared/notification-log";

let tmpRoot: string;
beforeAll(() => {
  tmpRoot = mkdtempSync(join(tmpdir(), "parked-row-requeue-"));
  const migDir = join(tmpRoot, "server", "db", "migrations");
  mkdirSync(migDir, { recursive: true });
  const real = join(import.meta.dir, "..", "db", "migrations");
  for (const f of readdirSync(real)) if (f.endsWith(".sql")) writeFileSync(join(migDir, f), readFileSync(join(real, f), "utf-8"));
  initDatabase(tmpRoot);
  configureNotificationRegistry({ announce: () => {}, announceSeen: () => {}, isTopicArchived: () => false });
});
afterAll(() => { try { closeDatabase(); } catch {} try { rmSync(tmpRoot, { recursive: true, force: true }); } catch {} });

const PID = "proj-notif";
function svc() {
  let n = 0; let clock = 0;
  return createTaskService(freshDb(), { now: () => new Date(Date.UTC(2026, 9, 3, 9, clock++)).toISOString(), uuid: () => `task-${Date.now()}-${++n}` });
}
const mv = (s: ReturnType<typeof svc>, taskId: string, status: string) =>
  s.update({ taskId, actor: "human", by: "user", patch: { status: status as never } });

describe("a parked card that is put back in the queue", () => {
  test("control: a card leaving review clears its bell row", () => {
    const s = svc();
    const t = s.create({ projectId: PID, text: "in review" });
    mv(s, t.id, "review");
    recordAndAnnounce({ kind: "task-review", title: "pronto", targetKind: "task", targetId: t.id, dedupeKey: taskReviewNotificationKey(t.id), source: "push" });
    expect(unseenSnapshot().unseenKeys).toContain(`task:${t.id}`);
    mv(s, t.id, "done");
    expect(unseenSnapshot().unseenKeys).not.toContain(`task:${t.id}`);
  });

  test("parked -> requeued: the task no longer waits, the row should be seen", () => {
    const s = svc();
    const t = s.create({ projectId: PID, text: "parcheggiata" });
    mv(s, t.id, "todo");
    // releaseAndEmit({requeue:false, parkState:'failed'}) -> backlog + task:parked -> push-trigger row
    s.release({ taskId: t.id, requeue: false, parkState: "failed", reason: "nessuna consegna" });
    recordAndAnnounce({ kind: "task-parked", title: "Task non consegnato", targetKind: "task", targetId: t.id, dedupeKey: taskParkedNotificationKey(t.id), source: "push" });
    console.log("after park:", s.get(t.id)!.task.status, s.get(t.id)!.task.dispatchState, JSON.stringify(unseenSnapshot()));
    // The person presses "Rimetti in coda" (PATCH status: todo).
    mv(s, t.id, "todo");
    console.log("after requeue:", s.get(t.id)!.task.status, JSON.stringify(unseenSnapshot()));
    expect(unseenSnapshot().unseenKeys).not.toContain(`task:${t.id}`);
  });
});
