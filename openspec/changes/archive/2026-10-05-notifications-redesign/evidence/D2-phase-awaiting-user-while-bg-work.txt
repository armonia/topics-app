/**
 * EVIDENCE (not a fix): the hook phase machine parks a session on
 * `awaiting-user` ("tocca a te", counted as attention + blue fill on the
 * client) when its turn ends with background work still running, unless that
 * work is a Monitor. Background Bash / Agent / Workflow / CronCreate never arm
 * `watching`.
 */
import { describe, expect, test } from "bun:test";
import { applyHook, makeInitialState, type HookPayload } from "./claude-session-state";

function turn(tool: string, input: Record<string, unknown>) {
  let s = makeInitialState("csid-1", "topic:x", 1000);
  const hook = (h: Partial<HookPayload>, t: number) => { s = applyHook(s, { session_id: "csid-1", ...h } as HookPayload, t); };
  hook({ hook_event_name: "UserPromptSubmit" }, 1001);
  hook({ hook_event_name: "PreToolUse", tool_name: tool, tool_input: input }, 1002);
  hook({ hook_event_name: "PostToolUse", tool_name: tool, tool_input: input }, 1003);
  hook({ hook_event_name: "Stop" }, 1004);
  return s.phase;
}

describe("the phase after a turn that leaves work running", () => {
  test("a Monitor keeps it 'watching' (control: the mechanism exists)", () => {
    expect(turn("Monitor", { command: "tail -f build.log" })).toBe("watching");
  });
  test("a background Bash", () => {
    const p = turn("Bash", { command: "bun test", run_in_background: true });
    console.log("Bash run_in_background -> phase", p);
    expect(p).toBe("watching");
  });
  test("a background Agent (3 verifiers)", () => {
    const p = turn("Agent", { description: "verifier", prompt: "...", run_in_background: true });
    console.log("Agent run_in_background -> phase", p);
    expect(p).toBe("watching");
  });
  test("a session cron", () => {
    const p = turn("CronCreate", { cron: "*/5 * * * *", prompt: "check CI" });
    console.log("CronCreate -> phase", p);
    expect(p).toBe("watching");
  });
});
