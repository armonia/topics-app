/**
 * EVIDENCE (not a fix): in a headless chat the built-in AskUserQuestion is
 * disallowed (providers/claude/args.ts HEADLESS_DISALLOWED_TOOLS) and every
 * question goes through `mcp__topics__ask_user_question`. The phase machine
 * only knows the built-in name, so a chat blocked on a question reads as
 * `tool-running` (working) and the "chat waits for you" push (PUSH-05) never
 * fires: push-triggers only reacts to `awaiting-approval`.
 */
import { describe, expect, test } from "bun:test";
import { applyHook, makeInitialState, type HookPayload } from "./claude-session-state";
import { configurePushTriggers, maybeSendPush } from "../push-triggers";
import type { NotificationRecordInput } from "../../shared/notification-log";

describe("a chat asking through the Topics MCP bridge", () => {
  test("is 'working' for the phase machine and sends no waiting push", () => {
    let s = makeInitialState("csid-ask", "topic:asks", 1000);
    const hook = (h: Partial<HookPayload>, t: number) => { s = applyHook(s, { session_id: "csid-ask", ...h } as HookPayload, t); };
    hook({ hook_event_name: "UserPromptSubmit" }, 1001);
    hook({ hook_event_name: "PreToolUse", tool_name: "mcp__topics__ask_user_question",
      tool_input: { questions: [{ question: "Procedo col merge?", options: [{ label: "Si" }, { label: "No" }] }] } }, 1002);
    const logged: NotificationRecordInput[] = [];
    configurePushTriggers({ getTopicName: () => "Merge", isTopicSilenced: () => false, recordNotification: (i) => { logged.push(i); },
      topicIdForSessionKey: () => "t-asks" });
    maybeSendPush({ type: "session:state", sessionKey: "topic:asks", state: s });
    console.log("phase =", s.phase, " pendingApproval =", JSON.stringify(s.pendingApproval), " rows =", JSON.stringify(logged));
    expect(s.phase).toBe("awaiting-approval");
    expect(logged.map((r) => r.kind)).toEqual(["session"]);
  });
});
