const SEEDED = HISTORY_FIRST_PAGE * 3;
const ROW = /Seeded message #(\d{3})/;
const seededText = (n: number) => `Seeded message #${String(n).padStart(3, "0")}`;
const REPLY_PROSE = "The agent explains what it did, quotes a path, and moves on to the next step. ".repeat(50);

async function sessionKeyOf(request: APIRequestContext, topicId: string): Promise<string> {
  const res = await request.get(`${E2E_BASE}/api/topics`);
  const { topics } = (await res.json()) as { topics: Record<string, { sessionKey: string }> };
  return topics[topicId]!.sessionKey;
}

type Frame = { t: number; state: string | null; rows: Record<string, number>; vis: string[]; inView: number; st: number; divider: number | null };

async function armFrameProbe(page: Page): Promise<void> {
  await page.evaluate(({ sel, rowPattern }) => {
    const w = window as unknown as { __frames: unknown[]; __probeStop?: boolean };
    w.__frames = [];
    w.__probeStop = false;
    const pattern = new RegExp(rowPattern);
    const MODE = (window as unknown as { __sampleMode?: string }).__sampleMode ?? "raf";
    const ch = new MessageChannel();
    ch.port1.onmessage = () => sample();
    // "ro": read after the frame's layout and the app's own ResizeObservers (created before this one), before paint.
    const sentinel = document.createElement("div");
    sentinel.style.cssText = "position:fixed;left:0;top:0;width:1px;height:1px;opacity:0;pointer-events:none";
    document.body.appendChild(sentinel);
    let frameNo = 0;
    let sampledFrame = -1;
    const ro = new ResizeObserver(() => { if (sampledFrame !== frameNo) { sampledFrame = frameNo; sample(); } });
    if (MODE === "ro") ro.observe(sentinel);
    const tick = () => {
      if (w.__probeStop) { ro.disconnect(); sentinel.remove(); return; }
      frameNo++;
      if (MODE === "post") ch.port2.postMessage(0);
      else if (MODE === "ro") sentinel.style.width = `${(frameNo % 2) + 1}px`;
      else sample();
      requestAnimationFrame(tick);
    };
    const sample = () => {
      if (w.__probeStop) return;
      const scroller = [...document.querySelectorAll(sel.replace(":visible", ""))].find((el) => (el as HTMLElement).offsetParent !== null) as HTMLElement | undefined;
      if (scroller) {
        const view = scroller.getBoundingClientRect();
        const rows: Record<string, number> = {};
        const vis: string[] = [];
        let inView = 0;
        for (const row of scroller.querySelectorAll("[data-index]")) {
          const box = row.getBoundingClientRect();
          const isIn = !(box.bottom <= view.top || box.top >= view.bottom);
          if (isIn) inView++;
          for (const m of row.querySelectorAll('[data-testid="chat-message"]')) {
            const n = pattern.exec(m.textContent || "")?.[1];
            if (n) { rows[n] = Math.round((m.getBoundingClientRect().top - view.top) * 10) / 10; if (isIn) vis.push(n); }
          }
        }
        const d = scroller.querySelector('[data-testid="chat-load-older"]');
        w.__frames.push({ t: performance.now(), state: scroller.getAttribute("data-history"), rows, vis, inView, st: Math.round(scroller.scrollTop), divider: d ? Math.round(d.getBoundingClientRect().height) : null });
      }
    };
    requestAnimationFrame(tick);
  }, { sel: SCROLLER, rowPattern: ROW.source });
}

async function stopProbe(page: Page): Promise<Frame[]> {
  return await page.evaluate(() => {
    const w = window as unknown as { __frames: Frame[]; __probeStop?: boolean };
    w.__probeStop = true;
    return w.__frames;
  });
}

async function stillView(page: Page): Promise<number> {
  return await page.evaluate(async (sel) => {
    const scroller = [...document.querySelectorAll(sel.replace(":visible", ""))].find((el) => (el as HTMLElement).offsetParent !== null) as HTMLElement;
    let last = -1;
    let still = 0;
    for (let i = 0; i < 120 && still < 4; i++) {
      await new Promise((r) => requestAnimationFrame(() => r(null)));
      const st = scroller.scrollTop;
      still = st === last ? still + 1 : 0;
      last = st;
    }
    return scroller.scrollTop;
  }, SCROLLER);
}

/** Rows in view in `a`: how far each moved by `b` (rendered there, in view or overscan). null: none still rendered. */
function moved(a: Frame, b: Frame): number | null {
  const common = a.vis.filter((k) => k in b.rows);
  if (common.length === 0) return null;
  return Math.max(...common.map((k) => Math.abs(b.rows[k] - a.rows[k])));
}

test.describe("probe 269c gestures", () => {
  test.describe.configure({ timeout: 180_000 });
  test.use({ viewport: { width: 1280, height: 720 } });
  test.use({ serviceWorkers: "block" });
  let longTopic: { id: string; name: string };
  let longKey = "";

  test.beforeAll(async ({ request }) => {
    test.setTimeout(180_000);
    longTopic = await createTopic(request, `probe269c-${Date.now()}`);
    longKey = await sessionKeyOf(request, longTopic.id);
    for (let i = 1; i <= SEEDED; i++) {
      const user = i % 2 === 1;
      await seedMessage(request, { sessionKey: longKey, role: user ? "user" : "assistant", content: user ? `${seededText(i)}\n\nA short question.` : `${seededText(i)}\n\n${REPLY_PROSE}` });
    }
  });
  test.afterAll(async ({ request }) => { if (longTopic) await deleteTopic(request, longTopic.id).catch(() => {}); });

  async function openOnTail(page: Page, request: APIRequestContext, delayMs = 0) {
    await resetPaneStore(request, [longTopic.id]);
    let older = 0;
    await page.route(`**/api/history/${encodeURIComponent(longKey)}*`, async (route) => {
      const body = route.request().postData() || "";
      if (body.includes('"before"')) { older++; if (delayMs) await new Promise((r) => setTimeout(r, delayMs)); }
      await route.continue();
    });
    await page.goto("/");
    await page.getByTestId(`pane-tab-${longTopic.id}`).click();
    const list = page.locator(SCROLLER);
    await expect(list.getByText(seededText(SEEDED))).toBeVisible({ timeout: 15000 });
    await expect(list).toHaveAttribute("data-history", "partial");
    return { list, older: () => older };
  }

  for (const delay of [0, 1500]) {
    test(`fast reader: Home from the bottom, older answer delayed ${delay} ms - do the rows being read move at the merge?`, async ({ page, request }) => {
      const { list, older } = await openOnTail(page, request, delay);
      await page.waitForTimeout(3000); // past the opening window
      await list.getByText(seededText(SEEDED - 1)).click();
      await stillView(page);
      await page.waitForTimeout(600);
      await armFrameProbe(page);
      const presses: string[] = [];
      for (let i = 0; i < 6; i++) {
        await page.keyboard.press("Home");
        await stillView(page);
        await page.waitForTimeout(delay + 700);
        await stillView(page);
        const stNow = await list.evaluate((el) => el.scrollTop);
        const state = await list.getAttribute("data-history");
        presses.push(`#${i + 1}:st=${stNow}:${state}`);
        if (state === "complete") break;
      }
      const fr = await page.evaluate(() => (window as unknown as { __frames: { st: number; state: string }[] }).__frames.map((f) => `${f.st}${f.state === "partial" ? "" : f.state[0]}`).join(" "));
      console.log(`[home:${delay}] presses=${presses.join(" ")} sts=${fr.slice(0, 3000)}`);
      await expect(list).toHaveAttribute("data-history", "complete", { timeout: 15000 });
      for (let i = 0; i < 6; i++) await stillView(page);
      const frames = await stopProbe(page);
      const at = frames.findIndex((f) => f.state === "complete");
      // frames from the first one at the top (st<=0) until the merge
      const topAt = frames.findIndex((f) => f.st <= 0);
      const after = frames.slice(at, at + 20).map((f) => moved(frames[at - 1], f));
      const steps = frames.slice(at, at + 20).map((f, i) => moved(frames[at + i - 1], f));
      const blank = frames.slice(Math.max(0, at - 2)).filter((f) => f.inView === 0).length;
      console.log(`[home:${delay}] older=${older()} frames=${frames.length} topAt=${topAt} mergeAt=${at} idleAtTopFrames=${at - topAt} before=${JSON.stringify({ st: frames[at - 1]?.st, rows: frames[at - 1]?.rows, divider: frames[at - 1]?.divider })} at=${JSON.stringify({ st: frames[at]?.st, rows: frames[at]?.rows, divider: frames[at]?.divider })} cumulative=${JSON.stringify(after)} perFrame=${JSON.stringify(steps)} blank=${blank}`);
      expect(older()).toBe(1);
      for (const m of after) expect(m ?? Infinity, "rows read at the top do not move at the merge").toBeLessThanOrEqual(1);
      expect(blank).toBe(0);
    });
  }

  for (const mode of ["raf", "post", "ro"] as const) for (const key of ["PageUp", "Shift+Space", "Home"] as const) {
    test(`keyboard ${key} across the merge (${mode}): do the rows move by more than the step?`, async ({ page, request }) => {
      await page.addInitScript((m) => { (window as unknown as { __sampleMode: string }).__sampleMode = m; }, mode);
      const { list, older } = await openOnTail(page, request);
      await page.waitForTimeout(3000);
      const box = (await list.boundingBox())!;
      await page.mouse.click(box.x + box.width / 2, box.y + box.height / 2);
      await stillView(page);
      if (key === "ArrowUp") {
        // Placed without a gesture just above the stage band: 40 px steps from the bottom take too long.
        const screen = await list.evaluate((el) => (el as HTMLElement).clientHeight);
        const target = Math.round(screen * 2.6);
        let st = await stillView(page);
        for (let i = 0; i < 40 && Math.abs(st - target) > 30; i++) {
          await list.evaluate((el, top) => { (el as HTMLElement).scrollTop = top; }, target);
          st = await stillView(page);
        }
      }
      await page.waitForTimeout(600);
      await armFrameProbe(page);
      let presses = 0;
      for (let i = 0; i < 400 && (await list.getAttribute("data-history")) !== "complete"; i++) {
        await page.waitForTimeout(450);
        await page.keyboard.press(key);
        presses++;
        await stillView(page);
      }
      for (let i = 0; i < 6; i++) await stillView(page);
      const frames = await stopProbe(page);
      await expect(list).toHaveAttribute("data-history", "complete");
      const at = frames.findIndex((f) => f.state === "complete");
      let landed = at;
      while (landed + 1 < frames.length && frames[landed + 1].st !== frames[landed].st) landed++;
      // the step: from the last still frame before the merge's press to the landing, in scrollTop terms is meaningless across a merge;
      // so measure the step of the press BEFORE the merge press as reference
      const across = moved(frames[at - 1], frames[landed]);
      const perFrame = frames.slice(at - 3, landed + 8).map((f, i) => (i === 0 ? 0 : moved(frames[at - 3 + i - 1], f)));
      const sts = frames.slice(Math.max(0, at - 30), landed + 8).map((f) => `${f.st}${f.state === "complete" ? "c" : f.state === "staged" ? "s" : ""}`).join(" ");
      const blank = frames.filter((f) => f.inView === 0).length;
      const visAround = frames.slice(at - 1, landed + 3).map((f) => `${f.st}:[${f.vis.join(",")}]@${f.vis.length ? f.rows[f.vis[0]] : "-"}`).join(" | ");
      // typical step of this key: median of the still-to-still deltas before the merge
      const stills: number[] = [];
      for (let i = 1; i < at; i++) if (frames[i].st === frames[i - 1].st && (stills.length === 0 || stills[stills.length - 1] !== frames[i].st)) stills.push(frames[i].st);
      const deltas = stills.slice(1).map((v, i) => stills[i] - v).filter((d) => d > 0).sort((a, b) => a - b);
      const step = deltas[Math.floor(deltas.length / 2)] ?? 0;
      console.log(`[key:${key}:${mode}] presses=${presses} older=${older()} mergeAt=${at} landed=${landed} typicalStep=${step} across=${across} perFrame=${JSON.stringify(perFrame)} blank=${blank} vis=${visAround} sts=${sts}`);
      expect(across, "a row read before the merge is still rendered after the landing").not.toBeNull();
      expect(across!, "across the merge the rows move by at most one step").toBeLessThanOrEqual(step + 1);
      for (const m of perFrame.slice(3)) expect(m ?? Infinity, "no frame after the merge moves the rows by more than a step").toBeLessThanOrEqual(step + 1);
      expect(blank).toBe(0);
    });
  }

  test("after a wheel-driven merge, does the chat announce the OLD rows as new messages?", async ({ page, request }) => {
    const { list, older } = await openOnTail(page, request);
    await page.waitForTimeout(3000);
    const box = (await list.boundingBox())!;
    await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2);
    for (let i = 0; i < 200 && (await list.getAttribute("data-history")) !== "complete"; i++) {
      await page.mouse.wheel(0, -300);
      await stillView(page);
    }
    await expect(list).toHaveAttribute("data-history", "complete");
    await page.waitForTimeout(800);
    const pill = await page.getByTestId("scroll-to-bottom").first().innerText().catch(() => "<none>");
    const banner = await page.getByText("New messages").count();
    console.log(`[banner] older=${older()} scrollToBottom=${JSON.stringify(pill)} newMessagesBanner=${banner}`);
    expect(pill.trim(), "no new message arrived: the pill must not count the old rows").not.toMatch(/\d/);
    expect(banner).toBe(0);
  });

  // Oracle for a press left held: after the action and > 500 ms of nothing, a NON-reader move of the list
  // up by 100 px (what a Virtuoso re-measure looks like to `onScroll`) must not ask for the rest.
  for (const action of ["none", "scrollbar-drag", "right-click", "text-drag", "click-content"] as const) {
    test(`held press oracle: ${action}`, async ({ page, request }) => {
      const { list, older } = await openOnTail(page, request);
      await page.waitForTimeout(3000);
      const screen = await list.evaluate((el) => (el as HTMLElement).clientHeight);
      // Placed without a gesture inside the band where heading up stages.
      const target = Math.round(screen * (OLDER_STAGE_SCREENS - 2));
      let st = await stillView(page);
      for (let i = 0; i < 40 && Math.abs(st - target) > 30; i++) {
        await list.evaluate((el, top) => { (el as HTMLElement).scrollTop = top; }, target);
        st = await stillView(page);
      }
      await page.waitForTimeout(800);
      expect(older(), "placing the list fetched nothing").toBe(0);
      await page.evaluate(() => {
        const w = window as unknown as { __pev: string[] };
        w.__pev = [];
        for (const t of ["pointerdown", "pointerup", "pointercancel", "mousedown", "mouseup", "dragstart", "dragend", "contextmenu", "blur"]) {
          window.addEventListener(t, (e) => w.__pev.push(`${t}:${(e.target as Element)?.nodeName ?? "?"}`), true);
        }
      });
      const geo = await list.evaluate((el) => {
        const r = el.getBoundingClientRect();
        const e = el as HTMLElement;
        return { x: r.x, y: r.y, w: r.width, h: r.height, sbw: e.offsetWidth - e.clientWidth, st: e.scrollTop, sh: e.scrollHeight, ch: e.clientHeight };
      });
      if (action === "scrollbar-drag") {
        const thumbY = geo.y + (geo.ch * geo.st) / geo.sh + Math.max(8, (geo.ch * geo.ch) / geo.sh / 2);
        const x = geo.x + geo.w - Math.max(3, geo.sbw / 2);
        await page.mouse.move(x, thumbY);
        await page.mouse.down();
        await page.mouse.move(x, thumbY + 4, { steps: 4 });
        await page.waitForTimeout(500);
        await page.mouse.up();
      } else if (action === "right-click") {
        await page.mouse.click(geo.x + geo.w / 2, geo.y + geo.h / 2, { button: "right" });
        await page.keyboard.press("Escape");
      } else if (action === "text-drag") {
        const b = { x: geo.x + geo.w / 4, width: geo.w / 2 };
        const y = geo.y + geo.h / 2;
        await page.mouse.click(b.x + b.width / 2, y, { clickCount: 3 });
        await page.mouse.move(b.x + b.width / 2, y);
        await page.mouse.down();
        await page.mouse.move(b.x + b.width / 2 + 150, y + 60, { steps: 10 });
        await page.mouse.up();
      } else if (action === "click-content") {
        await page.mouse.click(geo.x + geo.w / 2, geo.y + geo.h / 2);
      }
      await stillView(page);
      await page.waitForTimeout(900);
      const olderAfterAction = older();
      const stBefore = await list.evaluate((el) => el.scrollTop);
      await list.evaluate((el) => { (el as HTMLElement).scrollTop -= 100; });
      await stillView(page);
      await page.waitForTimeout(800);
      const ev = await page.evaluate(() => (window as unknown as { __pev: string[] }).__pev);
      console.log(`[held:${action}] geo=${JSON.stringify(geo)} events=${ev.join(",")} olderAfterAction=${olderAfterAction} stBefore=${stBefore} olderAfterNonReaderMove=${older()} state=${await list.getAttribute("data-history")}`);
      expect(older() - olderAfterAction, "a move of the list not caused by the reader asked for the rest").toBe(0);
    });
  }
});
